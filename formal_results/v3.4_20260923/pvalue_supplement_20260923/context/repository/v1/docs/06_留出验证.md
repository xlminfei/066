**06：把一个物种的全部结果一起留出，检验未参与训练的物种**

完成binary路线的本页后，可继续[06A计算ROC/AUC](06A_HighLow_ROC_AUC.md)。06A复用这里的拟合；本页的ELPD计算及参照方式保持原样。

先执行05的读取和诊断，选择`OUTCOME`与`VARIANT`。本页支持普通物种分组和系统发育距离分块。每个候选模型使用相同折；不把同一物种的不同实验分散到训练和测试两侧。

先建立并保存折分配。没有信息的完整[0,1]区间不会用来制造测试证据。M3编码字典继续使用已经冻结的无结局物种面板，不根据留出结果重新编码。

<!-- R_BLOCK:06_FOLDS -->
```r
CV_TYPE <- "species"       # 另一种选择是 phylo_distance。
CV_K <- 5L
CV_SEED <- 20260817L
CV_CHAINS <- CHAINS
CV_CORES <- CORES
CV_ITER <- ITER
CV_WARMUP <- WARMUP
CV_ADAPT_DELTA <- ADAPT_DELTA
CV_MAX_TREEDEPTH <- MAX_TREEDEPTH
if (!CV_TYPE %in% c("species","phylo_distance")) stop("CV_TYPE无效。")
if (OUTCOME=="binary") {
  cv_species<-prepared$binary_counts$Species[prepared$binary_counts$Trials>0]
} else {
  cv_species<-prepared$species$Species[prepared$species$Species %in%
    prepared$observations$Species[!prepared$observations$NoRatioInformation]]
}
if(CV_K<2 || CV_K>length(cv_species) || CV_K!=floor(CV_K))stop("CV_K必须在2和有信息物种数之间。")
set.seed(CV_SEED)
if(CV_TYPE=="species") {
  species_folds<-loo::kfold_split_random(K=CV_K,N=length(cv_species))
  names(species_folds)<-cv_species
} else {
  distances<-ape::cophenetic.phylo(prepared$tree)[cv_species,cv_species,drop=FALSE]
  species_folds<-stats::cutree(stats::hclust(stats::as.dist(distances),method="average"),k=CV_K)
  species_folds<-species_folds[cv_species]
}
if(!setequal(unique(as.integer(species_folds)),seq_len(CV_K)))stop("折分配不完整。")
fold_table<-data.frame(Species=cv_species,Fold=as.integer(species_folds),stringsAsFactors=FALSE)
fold_key<-digest::digest(list(OUTCOME,CV_TYPE,CV_K,CV_SEED,fold_table,prepared$input_hashes),algo="sha256")
CV_DIR<-file.path(RUN_DIR,"cv",paste0(OUTCOME,"_",CV_TYPE,"_",substr(fold_key,1,12)))
dir.create(CV_DIR,recursive=TRUE,showWarnings=FALSE)
if(!file.exists(file.path(CV_DIR,"folds.csv")))utils::write.csv(fold_table,file.path(CV_DIR,"folds.csv"),row.names=FALSE)
print(fold_table)
```

系统发育距离聚类不保证恰好是原树的单系类群，也不保证各折实验次数相同。结果应按实际分块方式命名。

每次选择一个模型，完成对应的拟合块和后面的共同诊断/评分块；再换下一个名称。正式比较需要十个模型全部完成。

分类路线使用brms现成的K-fold重拟合。不是对完整数据拟合的结果简单计算一个分数。

<!-- R_BLOCK:06_BINARY_CV -->
```r
if(OUTCOME!="binary")stop("这一块仅用于binary。")
if(file.exists(cv_file)) {
  cv_bundle<-readRDS(cv_file)
  if(!identical(cv_bundle$key,cv_key))stop("CV缓存身份不一致。")
} else {
  row_folds<-as.integer(species_folds[obj$request$training$Species])
  if(anyNA(row_folds))stop("有分类记录没有折编号。")
  cv_result<-brms::kfold(obj$fit,K=CV_K,folds=row_folds,joint="fold",save_fits=TRUE,
    seed=CV_SEED,cores=CV_CORES,chains=CV_CHAINS,iter=CV_ITER,warmup=CV_WARMUP,
    control=list(adapt_delta=CV_ADAPT_DELTA,max_treedepth=CV_MAX_TREEDEPTH))
  fold_fits<-vector("list",CV_K)
  heldout_rows<-vector("list",CV_K)
  for(k in seq_len(CV_K)) {
    fold_fits[[k]]<-cv_result$fits[[k,"fit"]]
    heldout_rows[[k]]<-as.integer(cv_result$fits[[k,"predicted"]])
    expected<-which(row_folds==k)
    if(!setequal(heldout_rows[[k]],expected))stop("实际留出行与指定折不一致。")
    if(obj$request$spec$Phylo) {
      if(!identical(levels(fold_fits[[k]]$data$Species),obj$species) ||
         !isTRUE(all.equal(fold_fits[[k]]$data2$A,obj$request$A,tolerance=0))) {
        stop("P模型的完整物种水平或协方差没有保留。")
      }
    }
  }
  predicted<-brms::kfold_predict(cv_result,method="posterior_epred",re_formula=NULL)
  row_ids<-as.integer(colnames(predicted$yrep))
  mean_draws_cv<-sweep(predicted$yrep,2L,obj$request$training$Trials[row_ids],FUN="/")
  cv_bundle<-list(key=cv_key,request=cv_request,outcome=OUTCOME,model=CV_MODEL,
    parent_key=obj$key,fold_fits=fold_fits,heldout_rows=heldout_rows,
    source=obj$request$training,result=cv_result,row_folds=row_folds,
    mean_draws_cv=mean_draws_cv,mean_row_ids=row_ids)
}
```

联合比率路线使用同一份Stan模型，逐折关闭留出物种的全部似然。需要本R会话已经执行03的模型字符串和编译块。下面的开关不会改变原输入，也不会把缺失结果补为0。

<!-- R_BLOCK:06_NATIVE_CV -->
```r
if(OUTCOME!="joint")stop("这一块仅用于joint。")
if(!exists("joint_compiled_code_hash") || !identical(obj$request$code_hash,joint_compiled_code_hash)) {
  stop("请用与主拟合相同的03模型说明编译；刚重拟合后先重新执行05读取。")
}
if(file.exists(cv_file)) {
  cv_bundle<-readRDS(cv_file)
  if(!identical(cv_bundle$key,cv_key))stop("CV缓存身份不一致。")
} else {
  fold_fits<-vector("list",CV_K);heldout_rows<-vector("list",CV_K)
  source<-obj$request$source_rows
  row_folds<-as.integer(species_folds[source$Species])
  for(k in seq_len(CV_K)) {
    d<-obj$request$data
    held<-which(!source$NoRatioInformation & !is.na(row_folds) & row_folds==k)
    if(!length(held))stop("出现空测试折。")
    heldout_rows[[k]]<-held
    d$train[held]<-0L
    held_species<-unique(source$Species[held])
    if(any(source$Species[d$train==1L] %in% held_species))stop("同一物种仍出现在训练侧。")
    fold_file<-file.path(CV_DIR,paste0(CV_MODEL,"_",substr(cv_key,1,16),"_fold",k,".rds"))
    data_key<-digest::digest(d,algo="sha256")
    if(file.exists(fold_file)) {
      cached_fold<-readRDS(fold_file)
      if(!identical(cached_fold$data_key,data_key))stop("折数据与缓存不一致。")
      fold_fits[[k]]<-cached_fold$fit
      next
    }
    set.seed(CV_SEED+k)
    iv<-vector("list",CV_CHAINS)
    for(ch in seq_along(iv)) {
      iv[[ch]]<-list(alpha=stats::rnorm(1,0,.3),beta=as.array(stats::rnorm(d$K,0,.1)),
        z_phylo=as.array(stats::rnorm(d$S*d$use_phylo,0,.2)),
        sd_phylo=as.array(rep(.3,d$use_phylo)),rho=stats::runif(1,.2,.6),
        log_phi_ratio=stats::rnorm(1,d$prior_log_phi_mean,.2),
        log_phi_count=as.array(stats::rnorm(d$use_beta_binomial,d$prior_log_phi_mean,.2)))
    }
    fold_fits[[k]]<-rstan::sampling(joint_compiled,data=d,init=iv,seed=CV_SEED+k,
      chains=CV_CHAINS,cores=CV_CORES,iter=CV_ITER,warmup=CV_WARMUP,
      control=list(adapt_delta=CV_ADAPT_DELTA,max_treedepth=CV_MAX_TREEDEPTH))
    if(length(fold_fits[[k]]@sim$samples)!=CV_CHAINS)stop("某折未获得全部链的样本。")
    saveRDS(list(data_key=data_key,fit=fold_fits[[k]]),fold_file)
  }
  cv_bundle<-list(key=cv_key,request=cv_request,outcome=OUTCOME,model=CV_MODEL,
    parent_key=obj$key,fold_fits=fold_fits,heldout_rows=heldout_rows,source=source,row_folds=row_folds)
}
```

两种路线都接着运行以下共同检查和评分。每折的联合分数是在同一后验抽样中先相加该折的对数似然，再进行后验积分；不能把各条边际预测概率的分数相加冒充整折联合分数。

<!-- R_BLOCK:06_SCORE -->
```r
fold_diagnostics<-list();fold_scores<-numeric(CV_K);prediction_rows<-list()
for(k in seq_len(CV_K)) {
  ff<-cv_bundle$fold_fits[[k]]
  sf<-if(OUTCOME=="binary")ff$fit else ff
  da<-posterior::as_draws_array(rstan::extract(sf,permuted=FALSE,inc_warmup=FALSE))
  vn<-posterior::variables(da)
  relevant<-if(OUTCOME=="binary")grepl("^(b_|sd_|r_|z_|phi$)",vn) else
    grepl("^(alpha$|beta\\[|z_phylo\\[|sd_phylo\\[|rho$|log_phi_ratio$|log_phi_count\\[|m\\[)",vn)
  ds<-as.data.frame(posterior::summarise_draws(posterior::subset_draws(da,variable=vn[relevant]),"rhat","ess_bulk","ess_tail"))
  sp<-rstan::get_sampler_params(sf,inc_warmup=FALSE)
  div<-0L;depth<-0L;bfmi<-numeric(length(sp))
  for(ch in seq_along(sp)) {
    div<-div+sum(sp[[ch]][,"divergent__"])
    depth<-depth+sum(sp[[ch]][,"treedepth__"]>=cv_bundle$request$sampling$max_treedepth)
    bfmi[ch]<-mean(diff(sp[[ch]][,"energy__"])^2)/stats::var(sp[[ch]][,"energy__"])
  }
  ok<-all(is.finite(c(ds$rhat,ds$ess_bulk,ds$ess_tail,bfmi))) &&
    max(ds$rhat)<MAX_RHAT && min(ds$ess_bulk)>=MIN_ESS && min(ds$ess_tail)>=MIN_ESS &&
    div==0 && depth==0 && min(bfmi)>.3
  fold_diagnostics[[k]]<-data.frame(Fold=k,Status=if(ok)"PASS"else"NEEDS_REVIEW",
    MaxRhat=max(ds$rhat),MinBulkESS=min(ds$ess_bulk),MinTailESS=min(ds$ess_tail),Divergences=div,TreeDepthHits=depth,MinEBFMI=min(bfmi))
  held<-cv_bundle$heldout_rows[[k]]
  if(OUTCOME=="binary") {
    if(nrow(cv_bundle$result$pointwise)!=CV_K)stop("包返回的评分单位不是整折。")
    fold_scores[k]<-cv_bundle$result$pointwise[k,"elpd_kfold"]
    pos<-match(held,cv_bundle$mean_row_ids)
    md<-cv_bundle$mean_draws_cv[,pos,drop=FALSE]
    ids<-cv_bundle$source$BinaryID[held]
  } else {
    la<-rstan::extract(sf,pars="log_lik",permuted=FALSE,inc_warmup=FALSE)[,,held,drop=FALSE]
    lm<-matrix(la,nrow=dim(la)[1]*dim(la)[2],ncol=dim(la)[3])
    if(any(!is.finite(lm)))stop("留出似然出现非有限值，不能形成可靠的数值评分。")
    fold_scores[k]<-matrixStats::logSumExp(rowSums(lm))-log(nrow(lm))
    ma<-rstan::extract(sf,pars="m",permuted=FALSE,inc_warmup=FALSE)
    mm<-matrix(ma,nrow=dim(ma)[1]*dim(ma)[2],ncol=dim(ma)[3])
    md<-mm[,match(cv_bundle$source$Species[held],obj$species),drop=FALSE]
    ids<-cv_bundle$source$RecordID[held]
  }
  test_species<-cv_bundle$source$Species[held]
  train_species<-cv_species[species_folds!=k]
  bp<-obj$blueprint
  ti<-match(train_species,prepared$species$Species)
  hi<-match(test_species,prepared$species$Species)
  level_seen<-rep(TRUE,length(hi))
  for(v in bp$variables)level_seen<-level_seen & prepared$species[[v]][hi] %in% prepared$species[[v]][ti]
  nb<-estimability::nonest.basis(cbind(1,bp$X[ti,,drop=FALSE]))
  estimable<-estimability::is.estble(cbind(1,bp$X[hi,,drop=FALSE]),nb)
  ci<-apply(md,2L,stats::quantile,probs=c(.025,.975),names=FALSE)
  prediction_rows[[k]]<-data.frame(RecordID=ids,Species=test_species,Fold=k,
    PosteriorMedian=apply(md,2L,stats::median),MeanLower95=ci[1,],MeanUpper95=ci[2,],
    LevelSeenInTraining=level_seen,FixedEffectEstimable=estimable,stringsAsFactors=FALSE)
}
cv_bundle$diagnostics<-do.call(rbind,fold_diagnostics)
cv_bundle$fold_scores<-data.frame(Fold=seq_len(CV_K),ELPD=fold_scores)
cv_bundle$predictions<-do.call(rbind,prediction_rows)
cv_bundle$status<-if(all(cv_bundle$diagnostics$Status=="PASS"))"PASS"else"NEEDS_REVIEW"
if(!file.exists(cv_file))saveRDS(cv_bundle,cv_file)
utils::write.csv(cv_bundle$diagnostics,file.path(CV_DIR,paste0(CV_MODEL,"_diagnostics.csv")),row.names=FALSE)
utils::write.csv(cv_bundle$predictions,file.path(CV_DIR,paste0(CV_MODEL,"_predictions.csv")),row.names=FALSE)
receipt<-data.frame(Model=CV_MODEL,Key=cv_key,ParentKey=obj$key,File=basename(cv_file),Status=cv_bundle$status,stringsAsFactors=FALSE)
ci_file<-file.path(CV_DIR,"cv_index.csv")
utils::write.table(receipt,ci_file,sep=",",row.names=FALSE,col.names=!file.exists(ci_file),append=file.exists(ci_file),quote=TRUE)
print(cv_bundle$diagnostics)
```

测试折中没有出现过的水平仍保留其预测和分数，明确标记，不删除难预测的记录后再比较模型。各折的后验来自不同训练集，因此不能把不同折的抽样当成同一次联合后验来计算跨折物种差异。

完成十个模型后汇总。分数越大表示该种留出设计下的预测表现越好；配对SE只是折间变异的近似表达，折数很少时不能过度解读。

<!-- R_BLOCK:06_COMPARE -->
```r
ix<-utils::read.csv(file.path(CV_DIR,"cv_index.csv"),stringsAsFactors=FALSE)
ix<-ix[!duplicated(ix$Model,fromLast=TRUE),,drop=FALSE]
if(!setequal(ix$Model,prepared$model_grid$Model))stop("当前留出设计的十个模型尚未齐全。")
scores<-matrix(NA_real_,CV_K,nrow(ix),dimnames=list(NULL,ix$Model))
for(j in seq_len(nrow(ix))) {
  v<-readRDS(file.path(CV_DIR,ix$File[j]))
  if(v$status!="PASS" || !identical(v$request$fold_key,fold_key))stop("有未通过诊断或不同折分配的CV结果。")
  if(!identical(v$parent_key,bundles[[ix$Model[j]]]$key))stop("主拟合选择已经变化，CV需要重新核对。")
  scores[,j]<-v$fold_scores$ELPD
}
total<-colSums(scores)
best<-names(which.max(total))
cv_comparison<-data.frame(Model=colnames(scores),ELPD=total,ELPD_Difference=NA_real_,PairedSE=NA_real_)
for(j in seq_len(ncol(scores))) {
  delta<-scores[,j]-scores[,best]
  cv_comparison$ELPD_Difference[j]<-sum(delta)
  cv_comparison$PairedSE[j]<-sqrt(CV_K*stats::var(delta))
}
cv_comparison<-cv_comparison[order(cv_comparison$ELPD,decreasing=TRUE),]
utils::write.csv(cv_comparison,file.path(CV_DIR,"cv_model_comparison.csv"),row.names=FALSE)
```

本页比较的是整折联合预测能力，不能直接与05逐行LOO的ELPD混排。是否能推广到远于固定物种面板的新对象，还需要相应资料；当前验证不会自动证明这种更广的推广能力。

整折的后验积分也受有限抽样误差影响，这与表中的折间配对SE不是同一种误差。分差很小或增加抽样后排序不稳定时，不应据此宣称某个模型明确更优。

如果某一折需要更长采样或更高的`adapt_delta`，只调整本页的`CV_`参数并重跑当前模型的CV即可；主拟合不必因此重跑。新的CV请求会使用新的缓存文件。联合模型已完成的折单独保存，中断后可以恢复。
