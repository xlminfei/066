**02：用真实High/Low实验次数，独立估计High倾向**

本分析不使用03的比率后验替代分类。`HighCount/Trials`统计的是实验结果被分为High的次数，不是个体行为响应的分子/分母。`PrHigh`表示纳入资料所代表实验背景下的High倾向。

当前分类口径为：单点/计数比率<50%归Low，>=50%归High；区间上界<=50%归Low，下界>=50%归High；严格跨越50%的区间不参与分类。尤其40–50%归Low，而精确50%归High。它们按报告形式使用这一明确约定。

02由能够明确分类的实验估计这一倾向。若无法分类或未被报告的实验与真实结果有关，不能把它直接解释为所有实验的无偏High比例；实验规模和观察终点也会影响这种分类倾向。

先选择一个模型。首次用`Null_U`，然后依次处理`Phylogeny_only_P`、`Site315_U`、`Site315_P`、`M1_U`、`M1_P`、`M2_U`、`M2_P`、`M3_U`、`M3_P`。每换一个名称，都重新执行本页的“准备”和“拟合保存”两块。

<!-- R_BLOCK:02_SELECT -->
```r
MODEL_NAME <- "Null_U"
VARIANT <- "primary"
BINARY_FAMILY <- "binomial"
```

下面构造当前模型。数字设计列的真实含义保存在`blueprint$columns`中，不能把`x01`等程序列名直接当作位点名称。M3只删除无变化的设计列，不删除其他共线列。

<!-- R_BLOCK:02_PREPARE -->
```r
prepared <- readRDS(file.path(RUN_DIR,"prepared.rds"))
if (!identical(prepared$version,MODEL_VERSION)) stop("准备结果版本不匹配。")
for (name in names(prepared$input_paths)) {
  if (!identical(digest::digest(file=prepared$input_paths[[name]],algo="sha256"),prepared$input_hashes[[name]])) {
    stop("原输入已更改；请创建新运行并重新准备。")
  }
}
if (!MODEL_NAME %in% prepared$model_grid$Model) stop("模型名称不在规定清单中。")
if (!BINARY_FAMILY %in% c("binomial","beta_binomial")) stop("分类分布名称无效。")
if (!grepl("^[A-Za-z0-9_-]+$",VARIANT)) stop("VARIANT只能使用字母、数字、短横线和下划线。")
spec <- prepared$model_grid[match(MODEL_NAME,prepared$model_grid$Model),,drop=FALSE]
blueprint <- prepared$blueprints[[paste0("binary_",spec$Group)]]
binary_rows <- prepared$binary_counts[prepared$binary_counts$Trials > 0,,drop=FALSE]
if (anyNA(binary_rows[,c("HighCount","LowCount","Trials")]) ||
    any(binary_rows$HighCount < 0 | binary_rows$LowCount < 0) ||
    any(binary_rows$HighCount + binary_rows$LowCount != binary_rows$Trials) ||
    any(as.matrix(binary_rows[,c("HighCount","LowCount","Trials")]) !=
        floor(as.matrix(binary_rows[,c("HighCount","LowCount","Trials")])))) {
  stop("分类计数表不满足真实整数次数及Trials=HighCount+LowCount的约定。")
}
species_index <- match(binary_rows$Species,prepared$species$Species)
binary_data <- cbind(binary_rows,as.data.frame(blueprint$X[species_index,,drop=FALSE]))
binary_data$Species <- factor(binary_data$Species,levels=prepared$species$Species)
rhs <- c("1",colnames(blueprint$X))
if (spec$Phylo) rhs <- c(rhs,"(1 | gr(Species, cov = A))")
formula_text <- paste("HighCount | trials(Trials) ~",paste(rhs,collapse=" + "))
formula_binary <- brms::bf(stats::as.formula(formula_text),center=FALSE)
family_binary <- if (BINARY_FAMILY == "binomial") stats::binomial(link="logit") else brms::beta_binomial(link="logit")
# center=FALSE时，截距属于b类、名称为Intercept；不能沿用旧class=Intercept写法。
priors_binary <- brms::set_prior(sprintf("normal(0, %.16g)",PRIOR_INTERCEPT_SD),class="b",coef="Intercept")
if (ncol(blueprint$X)) {
  priors_binary <- c(priors_binary,brms::set_prior(sprintf("normal(0, %.16g)",PRIOR_B_SD),class="b"))
}
if (spec$Phylo) {
  priors_binary <- c(priors_binary,brms::set_prior(sprintf("exponential(%.16g)",PRIOR_PHYLO_RATE),
    class="sd",group="Species",coef="Intercept"))
}
if (BINARY_FAMILY == "beta_binomial") {
  priors_binary <- c(priors_binary,brms::set_prior(sprintf("lognormal(%.16g, %.16g)",
    PRIOR_LOG_PHI_MEAN,PRIOR_LOG_PHI_SD),class="phi"))
}
binary_data2 <- if (spec$Phylo) list(A=prepared$A) else NULL
standata_binary <- brms::make_standata(formula_binary,data=binary_data,family=family_binary,
  prior=priors_binary,data2=binary_data2,drop_unused_levels=FALSE)
stancode_binary <- brms::make_stancode(formula_binary,data=binary_data,family=family_binary,
  prior=priors_binary,data2=binary_data2,drop_unused_levels=FALSE)
if (standata_binary$K != ncol(blueprint$X)+1L) stop("实际设计列数与预期不一致。")
sampling_settings <- list(seed=SEED,chains=CHAINS,cores=CORES,iter=ITER,warmup=WARMUP,
  adapt_delta=ADAPT_DELTA,max_treedepth=MAX_TREEDEPTH)
request <- list(version=MODEL_VERSION,outcome="binary",model=MODEL_NAME,variant=VARIANT,
  spec=spec,family=BINARY_FAMILY,formula_text=formula_text,
  code_hash=digest::digest(stancode_binary,algo="sha256",serialize=FALSE),
  source_hashes=prepared$input_hashes,training=binary_rows,
  source_map=prepared$observations[,c("RecordID","ExperimentID","Species","High","BinaryReason")],
  blueprint=blueprint,A=if(spec$Phylo)prepared$A else NULL,
  priors=as.data.frame(priors_binary),sampling=sampling_settings,
  runtime=list(R=R.version.string,platform=R.version$platform,packages=package_versions))
request_key <- digest::digest(request,algo="sha256")
relative_file <- file.path("fits","binary",VARIANT,paste0(MODEL_NAME,"_",substr(request_key,1,16),".rds"))
fit_file <- file.path(RUN_DIR,relative_file)
print(spec)
print(blueprint$columns)
print(priors_binary)
cat("物种观测数:",nrow(binary_rows),"；实验次数:",sum(binary_rows$Trials),"\n")
cat("设计秩:",blueprint$rank,"；含截距列数:",blueprint$parameter_columns,"\n")
```

下一块开始实际拟合。四条链完成后保存模型，但“已保存”不等于已经通过诊断。05会检查真实抽样结果。RDS文件名包含完整请求的哈希前缀，参数或输入改变会产生新文件，不覆盖旧拟合。

<!-- R_BLOCK:02_FIT_SAVE -->
```r
if (file.exists(fit_file)) {
  bundle <- readRDS(fit_file)
  if (!identical(bundle$key,request_key)) stop("缓存身份不一致，拒绝复用。")
  cat("复用本版匹配的分类拟合。\n")
} else {
  fit_binary <- brms::brm(formula=formula_binary,data=binary_data,family=family_binary,
    prior=priors_binary,data2=binary_data2,backend="rstan",drop_unused_levels=FALSE,
    save_pars=brms::save_pars(all=TRUE),seed=SEED,chains=CHAINS,cores=CORES,
    iter=ITER,warmup=WARMUP,control=list(adapt_delta=ADAPT_DELTA,max_treedepth=MAX_TREEDEPTH))
  if (length(fit_binary$fit@sim$samples) != CHAINS ||
      any(unlist(fit_binary$fit@sim$n_save)-unlist(fit_binary$fit@sim$warmup2) <= 0)) {
    stop("至少一条分类模型链没有有效后验样本；不保存为已拟合结果。")
  }
  bundle <- list(key=request_key,request=request,fit=fit_binary,blueprint=blueprint,
    training_data=binary_data,species=prepared$species$Species,
    finished_at=format(Sys.time(),tz="UTC",usetz=TRUE))
  dir.create(dirname(fit_file),recursive=TRUE,showWarnings=FALSE)
  saveRDS(bundle,fit_file)
}
# 索引只追加记录。05默认采用每个模型最近一次明确选择的本版拟合。
selection <- data.frame(Outcome="binary",Variant=VARIANT,Model=MODEL_NAME,Key=request_key,
  RelativeFile=relative_file,FileSHA256=digest::digest(file=fit_file,algo="sha256"),
  SelectedAt=format(Sys.time(),tz="UTC",usetz=TRUE),stringsAsFactors=FALSE)
index_file <- file.path(RUN_DIR,"fit_index.csv")
utils::write.table(selection,index_file,sep=",",row.names=FALSE,
  col.names=!file.exists(index_file),append=file.exists(index_file),quote=TRUE)
cat("模型已保存/选择。下一步检查05的诊断，再继续其他模型。\n")
```

你可以先对第一个模型运行05的诊断部分，再继续其余模型。模型比较和最终六模型输出需要所要求的模型齐全。已有未通过诊断的拟合不会因为索引中有记录就自动变成可靠结果。
