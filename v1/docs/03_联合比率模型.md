**03：让计数、点比率和真正区间共同估计物种期望比率**

本页必须在完成00、01之后使用。02与03是独立的分析；不把02的High概率当作03的比率。以下是标准二项分布、Beta分布和1膨胀混合结构的组合，没有自定义概率密度函数，也没有自制采样算法。

复核后明确增加了一项需要公开说明的简化：每个物种的整体期望为 `m`；在未知分母的比例分支中，精确100%点值的概率为 `rho * m`，其中 `rho` 为所有物种共享的端点贡献比例。内部Beta均值因此为 `(m-rho*m)/(1-rho*m)`。这个关系保证三类资料确实共享整体期望，同时只增加一项端点参数。共享 `rho` 是本方案的工作假设，不是现成分布自动保证的事实；后续必须检查它及先验是否明显影响结果。已知分母的count分支仍按计数分布计算端点概率，例如二项分布中全部个体均响应的概率是`m^Total`。

计数主分支使用二项分布；可选的beta-binomial计数敏感性使用自己的精度参数，不与未知分母比例的精度混为一谈。未知分母比例的离散程度不能解释成已经恢复了每条记录的分母。

先完整输入下面这个R字符串表达式。它只是把模型说明存进R对象，输入到结尾才结束，不会立刻开始拟合。

长字符串中的`data`声明01已经整理好的数据，`parameters`声明需要估计的量，`transformed parameters`把系数转换为物种期望与Beta形状，`model`规定先验和观测似然，`generated quantities`保存每条记录的评分量。你不需要逐个手填这些数组；本页后面的R准备块会自动构造。正式使用时只修改文中明确标出的设置，不改这个模型字符串里的公式。

```r
joint_model_code <- r"---(
data {
  int<lower=1> S;
  int<lower=0> K;
  int<lower=1> N;
  int<lower=0> N_count;
  int<lower=0> N_exact;
  int<lower=0> N_interval;
  matrix[S, K] X;
  cholesky_factor_cov[S] L_A;
  int<lower=0, upper=1> use_phylo;
  int<lower=0, upper=1> use_beta_binomial;
  array[N] int<lower=0, upper=1> train;
  array[N_count] int<lower=1, upper=S> species_count;
  array[N_count] int<lower=1, upper=N> row_count;
  array[N_count] int<lower=0> events;
  array[N_count] int<lower=1> total;
  array[N_exact] int<lower=1, upper=S> species_exact;
  array[N_exact] int<lower=1, upper=N> row_exact;
  vector<lower=0, upper=1>[N_exact] exact_ratio;
  array[N_interval] int<lower=1, upper=S> species_interval;
  array[N_interval] int<lower=1, upper=N> row_interval;
  vector<lower=0, upper=1>[N_interval] lower_ratio;
  vector<lower=0, upper=1>[N_interval] upper_ratio;
  real<lower=0> prior_intercept_sd;
  real<lower=0> prior_b_sd;
  real<lower=0> prior_phylo_rate;
  real<lower=0> prior_rho_a;
  real<lower=0> prior_rho_b;
  real prior_log_phi_mean;
  real<lower=0> prior_log_phi_sd;
}
transformed data {
  array[N] int seen = rep_array(0, N);
  for (i in 1:N_count) {
    seen[row_count[i]] += 1;
    if (events[i] > total[i]) reject("Events must not exceed Total");
  }
  for (i in 1:N_exact) {
    seen[row_exact[i]] += 1;
    if (exact_ratio[i] <= 0) reject("Unknown-N exact ratios must be strictly positive");
  }
  for (i in 1:N_interval) {
    seen[row_interval[i]] += 1;
    if (lower_ratio[i] >= upper_ratio[i]) reject("Intervals must have positive width");
  }
  for (i in 1:N) {
    if (seen[i] != 1) reject("Every source row must belong to exactly one observation type");
  }
}
parameters {
  real alpha;
  vector[K] beta;
  vector[use_phylo * S] z_phylo;
  vector<lower=0>[use_phylo] sd_phylo;
  real<lower=0, upper=1> rho;
  real log_phi_ratio;
  vector[use_beta_binomial] log_phi_count;
}
transformed parameters {
  vector[S] eta = rep_vector(alpha, S) + X * beta;
  vector[S] m;
  vector[S] endpoint_mass;
  vector[S] interior_mean;
  vector[S] log_endpoint;
  vector[S] log_continuous_weight;
  vector<lower=0>[S] shape_a;
  vector<lower=0>[S] shape_b;
  if (use_phylo == 1) eta += sd_phylo[1] * (L_A * z_phylo);
  m = inv_logit(eta);
  endpoint_mass = rho * m;
  for (s in 1:S) {
    real log_m = log_inv_logit(eta[s]);
    real log_one_minus_m = log1m_inv_logit(eta[s]);
    log_endpoint[s] = log(rho) + log_m;
    log_continuous_weight[s] = log1m_exp(log_endpoint[s]);
    interior_mean[s] = exp(log_m + log1m(rho) - log_continuous_weight[s]);
    shape_a[s] = exp(log_phi_ratio + log_m + log1m(rho) - log_continuous_weight[s]);
    shape_b[s] = exp(log_phi_ratio + log_one_minus_m - log_continuous_weight[s]);
  }
}
model {
  alpha ~ normal(0, prior_intercept_sd);
  beta ~ normal(0, prior_b_sd);
  z_phylo ~ std_normal();
  sd_phylo ~ exponential(prior_phylo_rate);
  rho ~ beta(prior_rho_a, prior_rho_b);
  log_phi_ratio ~ normal(prior_log_phi_mean, prior_log_phi_sd);
  log_phi_count ~ normal(prior_log_phi_mean, prior_log_phi_sd);
  for (i in 1:N_count) {
    if (train[row_count[i]] == 1) {
      int s = species_count[i];
      if (use_beta_binomial == 1) {
        real a_count = exp(log_phi_count[1] + log_inv_logit(eta[s]));
        real b_count = exp(log_phi_count[1] + log1m_inv_logit(eta[s]));
        target += beta_binomial_lpmf(events[i] | total[i], a_count, b_count);
      } else {
        target += binomial_logit_lpmf(events[i] | total[i], eta[s]);
      }
    }
  }
  for (i in 1:N_exact) {
    if (train[row_exact[i]] == 1) {
      int s = species_exact[i];
      if (exact_ratio[i] == 1) {
        target += log_endpoint[s];
      } else {
        target += log_continuous_weight[s]
          + beta_lpdf(exact_ratio[i] | shape_a[s], shape_b[s]);
      }
    }
  }
  for (i in 1:N_interval) {
    if (train[row_interval[i]] == 1) {
      int s = species_interval[i];
      real lo = lower_ratio[i];
      real hi = upper_ratio[i];
      if (lo == 0 && hi == 1) {
        target += 0;
      } else if (hi == 1) {
        target += log_sum_exp(log_endpoint[s],
          log_continuous_weight[s] + beta_lccdf(lo | shape_a[s], shape_b[s]));
      } else if (lo == 0) {
        target += log_continuous_weight[s]
          + beta_lcdf(hi | shape_a[s], shape_b[s]);
      } else if (lo >= interior_mean[s]) {
        target += log_continuous_weight[s] + log_diff_exp(
          beta_lccdf(lo | shape_a[s], shape_b[s]),
          beta_lccdf(hi | shape_a[s], shape_b[s]));
      } else {
        target += log_continuous_weight[s] + log_diff_exp(
          beta_lcdf(hi | shape_a[s], shape_b[s]),
          beta_lcdf(lo | shape_a[s], shape_b[s]));
      }
    }
  }
}
generated quantities {
  vector[N] log_lik;
  for (i in 1:N_count) {
    int s = species_count[i];
    if (use_beta_binomial == 1) {
      real a_count = exp(log_phi_count[1] + log_inv_logit(eta[s]));
      real b_count = exp(log_phi_count[1] + log1m_inv_logit(eta[s]));
      log_lik[row_count[i]] = beta_binomial_lpmf(events[i] | total[i], a_count, b_count);
    } else {
      log_lik[row_count[i]] = binomial_logit_lpmf(events[i] | total[i], eta[s]);
    }
  }
  for (i in 1:N_exact) {
    int s = species_exact[i];
    if (exact_ratio[i] == 1) {
      log_lik[row_exact[i]] = log_endpoint[s];
    } else {
      log_lik[row_exact[i]] = log_continuous_weight[s]
        + beta_lpdf(exact_ratio[i] | shape_a[s], shape_b[s]);
    }
  }
  for (i in 1:N_interval) {
    int s = species_interval[i];
    real lo = lower_ratio[i];
    real hi = upper_ratio[i];
    if (lo == 0 && hi == 1) {
      log_lik[row_interval[i]] = 0;
    } else if (hi == 1) {
      log_lik[row_interval[i]] = log_sum_exp(log_endpoint[s],
        log_continuous_weight[s] + beta_lccdf(lo | shape_a[s], shape_b[s]));
    } else if (lo == 0) {
      log_lik[row_interval[i]] = log_continuous_weight[s]
        + beta_lcdf(hi | shape_a[s], shape_b[s]);
    } else if (lo >= interior_mean[s]) {
      log_lik[row_interval[i]] = log_continuous_weight[s] + log_diff_exp(
        beta_lccdf(lo | shape_a[s], shape_b[s]),
        beta_lccdf(hi | shape_a[s], shape_b[s]));
    } else {
      log_lik[row_interval[i]] = log_continuous_weight[s] + log_diff_exp(
        beta_lcdf(hi | shape_a[s], shape_b[s]),
        beta_lcdf(lo | shape_a[s], shape_b[s]));
    }
  }
}
)---"
```

`train`是留出验证使用的开关。训练目标只计算开关为1的记录；所有记录的预测似然在抽样后的generated quantities中计算，避免极端留出区间的数值问题污染训练梯度。未知分母资料没有被放进计数数组，因此没有补造Total。模型输出的`m`已经包含100%端点贡献，07将直接汇总它，不汇总内部Beta均值，也不模拟未来实验来构造均值区间。
