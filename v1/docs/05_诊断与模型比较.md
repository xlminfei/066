**05：先检查抽样质量，再比较同一批资料上的模型**

本页第一部分可在只完成一个模型后使用。比较部分要求所选择的一条分析路线有完整的十个模型。`binary`与`joint`分别比较，不能把两种响应的ELPD混在一起排名。

先选择分析和版本，然后读取追加式索引中的最近一次明确选择。代码会检查RDS文件本身的哈希、请求身份和输入身份。

<!-- R_BLOCK:05_LOAD -->
```r
OUTCOME <- "binary"          # 分类为binary，联合比率为joint。
VARIANT <- "primary"
prepared <- readRDS(file.path(RUN_DIR,"prepared.rds"))
if(!identical(prepared$version,MODEL_VERSION))stop("准备结果使用了旧分类版本，请在新运行中重新执行01。")
for (name in names(prepared$input_paths)) {
  if (!identical(digest::digest(file=prepared$input_paths[[name]],algo="sha256"),prepared$input_hashes[[name]])) {
    stop("输入发生改变，请不要把旧拟合当作当前结果。")
  }
}
fit_index <- utils::read.csv(file.path(RUN_DIR,"fit_index.csv"),stringsAsFactors=FALSE,check.names=FALSE)
selected <- fit_index[fit_index$Outcome==OUTCOME & fit_index$Variant==VARIANT,,drop=FALSE]

if (!nrow(selected)) stop("这条路线/版本没有已选择的拟合。")
bundles <- vector("list",nrow(selected));names(bundles)<-selected$Model
for (i in seq_len(nrow(selected))) {
  file <- file.path(RUN_DIR,selected$RelativeFile[i])
  if (!identical(digest::digest(file=file,algo="sha256"),selected$FileSHA256[i])) stop("拟合文件内容与索引不一致。")
  obj <- readRDS(file)
  if(!identical(obj$request$version,MODEL_VERSION))stop("拟合来自旧运行版本，不能混入本版结果。")
  if (!identical(obj$key,selected$Key[i]) || !identical(digest::digest(obj$request,algo="sha256"),obj$key)) {
    stop("模型请求身份核查失败。")
  }
  if (!identical(obj$request$source_hashes,prepared$input_hashes)) stop("模型来自不同输入版本。")
  if (!identical(obj$species,prepared$species$Species) ||
      !identical(obj$blueprint,prepared$blueprints[[paste0(OUTCOME,"_",obj$request$spec$Group)]])) {
    stop("物种顺序或编码与当前准备结果不同。")
  }
  if(obj$request$spec$Phylo && !isTRUE(all.equal(obj$request$A,prepared$A,tolerance=0)))stop("系统发育矩阵与当前准备结果不同。")
  bundles[[i]] <- obj
}
observation_signatures<-character(length(bundles))
for(i in seq_along(bundles)) {

  observation_signatures[i]<-if(OUTCOME=="joint")paste(z$code_hash,z$count_family)else z$family
}
```

运行诊断。Rhat、有效样本量、发散、树深度和能量诊断分别回答不同问题；任何一项未满足，本版都不会将该拟合标为通过。阈值是数值检查的起点，不是对模型科学正确性的证明。

<!-- R_BLOCK:05_DIAGNOSTICS -->
```r
diagnostic_rows <- list()
parameter_rows <- list()
for (model in names(bundles)) {
  obj <- bundles[[model]]
  if (OUTCOME == "binary") {
    stan_fit <- obj$fit$fit
    draws_all <- brms::as_draws_array(obj$fit)
    vars <- posterior::variables(draws_all)
    keep <- grepl("^(b_|sd_|r_|phi$)",vars)
  } else {
    stan_fit <- obj$fit
    draws_all <- posterior::as_draws_array(rstan::extract(stan_fit,permuted=FALSE,inc_warmup=FALSE))
    vars <- posterior::variables(draws_all)
    keep <- grepl("^(alpha$|beta\\[|z_phylo\\[|sd_phylo\\[|rho$|log_phi_ratio$|log_phi_count\\[|m\\[)",vars)
  }
  if (!any(keep)) stop("没有找到应检查的参数。")
  draws <- posterior::subset_draws(draws_all,variable=vars[keep])
  ps <- as.data.frame(posterior::summarise_draws(draws,"mean","sd","rhat","ess_bulk","ess_tail","mcse_mean"))
  mat <- as.matrix(posterior::as_draws_matrix(draws))
  ci <- apply(mat,2L,stats::quantile,probs=c(.025,.975),names=FALSE)
  ps$Lower95 <- ci[1,match(ps$variable,colnames(mat))]
  ps$Upper95 <- ci[2,match(ps$variable,colnames(mat))]
  sampler <- rstan::get_sampler_params(stan_fit,inc_warmup=FALSE)
  divergences <- 0L;depth_hits <- 0L;ebfmi <- numeric(length(sampler))
  for (chain in seq_along(sampler)) {
    z <- sampler[[chain]]
    divergences <- divergences + sum(z[,"divergent__"])
    depth_hits <- depth_hits + sum(z[,"treedepth__"] >= obj$request$sampling$max_treedepth)
    ebfmi[chain] <- mean(diff(z[,"energy__"])^2)/stats::var(z[,"energy__"])
  }
  numbers <- c(ps$rhat,ps$ess_bulk,ps$ess_tail,ebfmi)
  numerical_ok <- all(is.finite(numbers)) && length(sampler)>=2L &&
    max(ps$rhat)<MAX_RHAT && min(ps$ess_bulk)>=MIN_ESS && min(ps$ess_tail)>=MIN_ESS &&
    divergences==0 && depth_hits==0 && min(ebfmi)>.3
  diagnostic_rows[[model]] <- data.frame(Outcome=OUTCOME,Variant=VARIANT,Model=model,Key=obj$key,
    Status=if(numerical_ok)"PASS"else"NEEDS_REVIEW",MaxRhat=max(ps$rhat),
    MinBulkESS=min(ps$ess_bulk),MinTailESS=min(ps$ess_tail),Divergences=divergences,
    TreeDepthHits=depth_hits,MinEBFMI=min(ebfmi),Rank=obj$blueprint$rank,
    DesignColumns=obj$blueprint$parameter_columns,stringsAsFactors=FALSE)
  ps$Outcome<-OUTCOME;ps$Variant<-VARIANT;ps$Model<-model
  ps$OriginalTerm <- ps$variable
  for (j in seq_len(nrow(obj$blueprint$columns))) {
    name <- if(OUTCOME=="binary")paste0("b_",obj$blueprint$columns$Column[j])else paste0("beta[",j,"]")
    ps$OriginalTerm[ps$variable==name] <- obj$blueprint$columns$Term[j]
  }
  ps$RankDeficientModel <- obj$blueprint$rank < obj$blueprint$parameter_columns
  parameter_rows[[model]] <- ps
}
diagnostics <- do.call(rbind,diagnostic_rows);rownames(diagnostics)<-NULL
parameter_summary <- do.call(rbind,parameter_rows);rownames(parameter_summary)<-NULL
diagnostic_file <- file.path(RUN_DIR,"results",paste0("diagnostics_",OUTCOME,"_",VARIANT,".csv"))
utils::write.csv(diagnostics,diagnostic_file,row.names=FALSE,na="")
utils::write.csv(parameter_summary,file.path(RUN_DIR,"results",paste0("parameter_summary_",OUTCOME,"_",VARIANT,".csv")),row.names=FALSE,na="")
print(diagnostics)
```

`NEEDS_REVIEW`时先看是哪项未通过。增加迭代有时有帮助，但不能修复数据重复、模型不合适或M3共线。M3即使数值诊断通过，也不能把共线系数解释成被数据分别识别的独立作用。

下面计算LOO。联合模型的`log_lik`按原实验记录保存；分类模型按物种汇总行保存。因此两种LOO的留出单位不同。对于联合模型，留出某一实验时仍可能有该物种的其他实验留在训练中；新物种的验证见06。

<!-- R_BLOCK:05_LOO -->
```r
if (any(diagnostics$Status!="PASS")) stop("先解决未通过的抽样诊断，再计算正式比较。")
loo_objects <- list();loo_status_rows <- list();loo_source_rows <- list()
comparison_identity <- NULL
for (model in names(bundles)) {
  obj<-bundles[[model]]
  if (OUTCOME=="binary") {
    identity <- obj$request$training[,c("BinaryID","Species","HighCount","LowCount","Trials"),drop=FALSE]
    ll_matrix <- brms::log_lik(obj$fit)
    draw_dimensions <- dim(brms::as_draws_array(obj$fit))
    if(nrow(ll_matrix)!=draw_dimensions[1]*draw_dimensions[2])stop("分类似然抽样数不匹配。")
    # brms返回抽样×观测矩阵；按其实际迭代/链顺序恢复数组。
    ll_array <- array(ll_matrix,dim=c(draw_dimensions[1],draw_dimensions[2],ncol(ll_matrix)))
    mean_draws <- brms::posterior_epred(obj$fit,re_formula=NULL)
    mean_draws <- sweep(mean_draws,2L,identity$Trials,FUN="/")
    source <- data.frame(RecordID=identity$BinaryID,Species=identity$Species,Type="binary_count",
      Events=identity$HighCount,Total=identity$Trials,Exact=identity$HighCount/identity$Trials,
      Lower=NA_real_,Upper=NA_real_,stringsAsFactors=FALSE)
  } else {
    rows <- which(!obj$request$source_rows$NoRatioInformation)
    identity <- obj$request$source_rows[rows,c("RecordID","Species","Type","Events","Total","Exact","Lower","Upper"),drop=FALSE]
    ll_all <- rstan::extract(obj$fit,pars="log_lik",permuted=FALSE,inc_warmup=FALSE)
    if (dim(ll_all)[3]!=nrow(obj$request$source_rows)) stop("log_lik与来源记录数量不一致。")
    ll_array <- ll_all[,,rows,drop=FALSE]
    m_all <- rstan::extract(obj$fit,pars="m",permuted=FALSE,inc_warmup=FALSE)
    m_matrix <- matrix(m_all,nrow=dim(m_all)[1]*dim(m_all)[2],ncol=dim(m_all)[3])
    mean_draws <- m_matrix[,match(identity$Species,obj$species),drop=FALSE]
    source <- identity
  }
  rownames(identity)<-NULL
  identity_key<-digest::digest(identity,algo="sha256")
  if (is.null(comparison_identity)) comparison_identity<-identity_key
  if (!identical(comparison_identity,identity_key)) stop("候选模型没有使用同一批、同一顺序和定义的响应。")
  if (length(dim(ll_array))!=3L || dim(ll_array)[3]!=nrow(identity)) stop("似然数组维度不匹配。")
  ll<-matrix(ll_array,nrow=dim(ll_array)[1]*dim(ll_array)[2],ncol=dim(ll_array)[3])
  if(any(!is.finite(ll)))stop("似然包含非有限数值，需要核查模型或数值计算。")
  chain_id<-rep(seq_len(dim(ll_array)[2]),each=dim(ll_array)[1])
  scaled_likelihood<-exp(sweep(ll,2L,apply(ll,2L,max),FUN="-"))
  r_eff<-loo::relative_eff(scaled_likelihood,chain_id=chain_id,cores=CORES)
  for(j in seq_len(ncol(ll))) if(length(unique(ll[,j]))==1L)r_eff[j]<-1
  if(any(!is.finite(r_eff)))stop("无法计算似然的链相关修正。")
  lo<-loo::loo(ll,r_eff=r_eff,save_psis=TRUE,cores=CORES)
  loo_objects[[model]]<-lo
  k<-loo::pareto_k_values(lo)
  k_limit<-min(.7,1-1/log10(nrow(ll)))
  good<-is.finite(k)&k<=k_limit
  loo_status_rows[[model]]<-data.frame(Model=model,MaxParetoK=max(k),KLimit=k_limit,
    ProblemRows=sum(!good),Status=if(all(good))"PASS"else"USE_GROUPED_CV",stringsAsFactors=FALSE)
  source$Model<-model;source$Outcome<-OUTCOME;source$Variant<-VARIANT
  source$ParetoK<-k;source$LOO_ExpectedValue<-NA_real_
  source$LOO_MeanLower95<-NA_real_;source$LOO_MeanUpper95<-NA_real_
  source$PredictionStatus<-"PSIS_not_validated_use_06"
  if(all(good)) {
    em<-loo::E_loo(mean_draws,psis_object=lo$psis_object,type="mean")
    eq<-loo::E_loo(mean_draws,psis_object=lo$psis_object,type="quantile",probs=c(.025,.975))
    prediction_good<-is.finite(em$pareto_k)&is.finite(eq$pareto_k)&em$pareto_k<=k_limit&eq$pareto_k<=k_limit
    source$LOO_ExpectedValue[prediction_good]<-em$value[prediction_good]
    source$LOO_MeanLower95[prediction_good]<-eq$value[1,prediction_good]
    source$LOO_MeanUpper95[prediction_good]<-eq$value[2,prediction_good]
    source$PredictionStatus[prediction_good]<-"PSIS_checked"
  }
  loo_source_rows[[model]]<-source
}
loo_status<-do.call(rbind,loo_status_rows);rownames(loo_status)<-NULL
loo_sources<-do.call(rbind,loo_source_rows);rownames(loo_sources)<-NULL
saveRDS(list(objects=loo_objects,status=loo_status,selection=selected,identity=comparison_identity),
  file.path(RUN_DIR,"results",paste0("loo_objects_",OUTCOME,"_",VARIANT,".rds")))
utils::write.csv(loo_status,file.path(RUN_DIR,"results",paste0("loo_status_",OUTCOME,"_",VARIANT,".csv")),row.names=FALSE,na="")
utils::write.csv(loo_sources,file.path(RUN_DIR,"results",paste0("loo_source_means_",OUTCOME,"_",VARIANT,".csv")),row.names=FALSE,na="")
print(loo_status)
```

如果PSIS不可靠，本版会保留问题并转用06的真实留出验证，不把不可靠权重产生的预测当作已修复结果。LOO来源表中的区间是留一条件下期望量的后验区间；分类时的期望量是High概率，联合时是整体期望比率。

只有模型齐全、抽样和PSIS检查通过时，才运行正式配对比较。差值的标准误由`loo_compare()`按相同观测的配对贡献计算，不从两个模型各自的标准误猜算。
