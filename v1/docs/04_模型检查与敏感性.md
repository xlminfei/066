<!-- R_BLOCK:04_PPC -->
```r
ppc_sets<-list()
if(OUTCOME=="binary") {
  yp<-brms::posterior_predict(obj$fit,ndraws=500)
  yp<-sweep(yp,2L,obj$request$training$Trials,FUN="/")
  yy<-obj$request$training$HighCount/obj$request$training$Trials
  ppc_sets$HighFraction<-list(observed=yy,replicated=yp)
} else {
  da<-rstan::extract(obj$fit,permuted=FALSE,inc_warmup=FALSE)
  dm<-as.matrix(posterior::as_draws_matrix(posterior::as_draws_array(da)))
  use<-sample.int(nrow(dm),min(500L,nrow(dm)))
  obs<-obj$request$source_rows
  sid<-match(obs$Species,obj$species)
  for(kind in c("count","exact")) {
    rows<-which(obs$Type==kind)
    if(!length(rows))next
    reps<-matrix(NA_real_,length(use),length(rows))
    for(a in seq_along(use)) {
      draw<-stats::setNames(as.numeric(dm[use[a],,drop=FALSE]),colnames(dm))
      m<-draw[paste0("m[",sid[rows],"]")]
      if(kind=="count") {
        if(obj$request$count_family=="beta_binomial") {
          pc<-exp(draw["log_phi_count[1]"])
          prob<-stats::rbeta(length(rows),m*pc,(1-m)*pc)
        } else prob<-m
        reps[a,]<-stats::rbinom(length(rows),obs$Total[rows],prob)/obs$Total[rows]
      } else {
        w<-draw["rho"]*m
        sa<-draw[paste0("shape_a[",sid[rows],"]")]
        sb<-draw[paste0("shape_b[",sid[rows],"]")]
        at_one<-stats::rbinom(length(rows),1,w)
        reps[a,]<-ifelse(at_one==1,1,stats::rbeta(length(rows),sa,sb))
      }
    }
    yy<-if(kind=="count")obs$Events[rows]/obs$Total[rows]else obs$Exact[rows]
    ppc_sets[[kind]]<-list(observed=yy,replicated=reps)
  }
  ir<-which(obs$Type=="interval")
  if(length(ir)) {
    probabilities<-exp(dm[,paste0("log_lik[",ir,"]"),drop=FALSE])
    interval_check<-data.frame(RecordID=obs$RecordID[ir],Species=obs$Species[ir],
      Lower=obs$Lower[ir],Upper=obs$Upper[ir],
      PosteriorMeanIntervalProbability=colMeans(probabilities),stringsAsFactors=FALSE)
    utils::write.csv(interval_check,file.path(RUN_DIR,"results",paste0("interval_check_",CHECK_MODEL,"_",VARIANT,".csv")),row.names=FALSE)
    print(interval_check)
  }
}
ppc_summary<-list()
for(kind in names(ppc_sets)) {
  s<-ppc_sets[[kind]]
  if(any(!is.finite(s$replicated)))stop("后验模拟出现非有限值，先检查参数与记录的对应关系。")
  replicated_means<-rowMeans(s$replicated)
  q<-stats::quantile(replicated_means,c(.025,.5,.975),names=FALSE)
  ppc_summary[[kind]]<-data.frame(Subset=kind,ObservedMean=mean(s$observed),
    ReplicatedMeanLower=q[1],ReplicatedMeanMedian=q[2],ReplicatedMeanUpper=q[3],
    ObservedZeroFraction=mean(s$observed==0),ReplicatedZeroFraction=mean(s$replicated==0),
    ObservedOneFraction=mean(s$observed==1),ReplicatedOneFraction=mean(s$replicated==1))
  plot_file<-file.path(RUN_DIR,"results",paste0("ppc_",OUTCOME,"_",CHECK_MODEL,"_",VARIANT,"_",kind,".pdf"))
  grDevices::pdf(plot_file)
  print(bayesplot::ppc_stat(s$observed,s$replicated,stat="mean"))
  if(length(s$observed)>1L)print(bayesplot::ppc_stat(s$observed,s$replicated,stat="sd"))
  print(bayesplot::ppc_ecdf_overlay(s$observed,s$replicated[seq_len(min(50L,nrow(s$replicated))),,drop=FALSE]))
  grDevices::dev.off()
}
ppc_summary<-do.call(rbind,ppc_summary)
utils::write.csv(ppc_summary,file.path(RUN_DIR,"results",paste0("ppc_summary_",OUTCOME,"_",CHECK_MODEL,"_",VARIANT,".csv")),row.names=FALSE)
print(ppc_summary)
```

PDF同时检查均值、标准差和经验分布，CSV补充0与1的出现比例；仅均值接近不足以说明拟合良好。区间很宽时，落入区间的概率自然较高；不能把它当成“这条资料正确的概率”。后验预测检查使用了拟合资料，也不是独立的外部验证。明显不合时应重新检查观测定义与模型，不仅增加采样次数。

建议预先指定以下少量敏感性设置。它们不覆盖主模型，分别使用新的VARIANT。每设置一次，回到02或03，只运行准备和拟合保存块；不要重新运行选择块把VARIANT重置为primary。

| 检查 | 设置 | 至少复核哪些结构 |
|---|---|---|
| M3更强收缩 | b先验SD=0.25 | M3_U、M3_P |
| M3较弱收缩 | b先验SD=1 | M3_U、M3_P |
| 端点先验 | rho先验Beta(2,2) | 联合M1_U/P与M3_U/P |
| 额外计数变异 | beta-binomial | 对有足够重复计数信息的主要结构检查 |

以下四块分别选择，不能连续输入后只运行最后一组。每次切换检查前先运行本页末尾的`04_RESET`恢复主设置，再执行所选设置块，防止上一项检查的计数分布或端点先验意外沿用。

每个新增拟合也要在05通过诊断。额外离散参数并不会恢复未知分母或消除研究背景差异；在Trials几乎都为1时，分类beta-binomial的额外参数缺少识别信息。

以下代码把某个敏感性拟合与同一路线、同一模型的主拟合比较，直接查看期望量的后验中位数和区间改变了多少。不跨资料集混比LOO，也不把两套后验平均。

<!-- R_BLOCK:04_COMPARE_SENSITIVITY -->
```r
SENSITIVITY_VARIANT <- "b_sd_025"
SENSITIVITY_MODEL <- "M3_P"
ix<-utils::read.csv(file.path(RUN_DIR,"fit_index.csv"),stringsAsFactors=FALSE)
tables<-list()
for(v in c("primary",SENSITIVITY_VARIANT)) {
  r<-ix[ix$Outcome==OUTCOME & ix$Variant==v & ix$Model==SENSITIVITY_MODEL,,drop=FALSE]
  if(!nrow(r))stop(paste("缺少",v,"拟合。"))
  r<-r[nrow(r),,drop=FALSE]
  zfile<-file.path(RUN_DIR,r$RelativeFile)
  if(!identical(digest::digest(file=zfile,algo="sha256"),r$FileSHA256))stop("敏感性拟合文件哈希改变。")
  z<-readRDS(zfile)
  if(!identical(z$request$version,MODEL_VERSION))stop("敏感性拟合来自旧运行版本，不能混用。")
  if(!identical(z$key,r$Key) || !identical(z$key,digest::digest(z$request,algo="sha256")))stop("敏感性拟合身份不一致。")
  dg<-utils::read.csv(file.path(RUN_DIR,"results",paste0("diagnostics_",OUTCOME,"_",v,".csv")),stringsAsFactors=FALSE)
  if(!any(dg$Key==z$key & dg$Status=="PASS"))stop("先完成该敏感性拟合的05诊断。")
  if(!identical(z$request$source_hashes,prepared$input_hashes))stop("敏感性结果的数据版本不同。")
  if(OUTCOME=="binary") {
    nd<-cbind(data.frame(Species=factor(z$species,levels=z$species),Trials=1L),as.data.frame(z$blueprint$X))
    m<-brms::posterior_epred(z$fit,newdata=nd,re_formula=NULL,allow_new_levels=FALSE)
  } else {
    a<-rstan::extract(z$fit,pars="m",permuted=FALSE,inc_warmup=FALSE)
    m<-matrix(a,nrow=dim(a)[1]*dim(a)[2],ncol=dim(a)[3])
  }
  q<-apply(m,2L,stats::quantile,probs=c(.025,.5,.975),names=FALSE)
  tables[[v]]<-data.frame(Species=z$species,Lower=q[1,],Median=q[2,],Upper=q[3,],Supported=z$blueprint$supported)
}
base<-tables$primary;alt<-tables[[SENSITIVITY_VARIANT]]
if(!identical(base$Species,alt$Species))stop("物种顺序不同。")
comparison<-data.frame(Species=base$Species,PrimaryMedian=base$Median,AlternativeMedian=alt$Median,
  MedianShift=alt$Median-base$Median,PrimaryLower=base$Lower,PrimaryUpper=base$Upper,
  AlternativeLower=alt$Lower,AlternativeUpper=alt$Upper,Supported=base$Supported & alt$Supported)
comparison[!comparison$Supported,setdiff(names(comparison),c("Species","Supported"))]<-NA_real_
utils::write.csv(comparison,file.path(RUN_DIR,"results",paste0("sensitivity_",OUTCOME,"_",SENSITIVITY_MODEL,"_",SENSITIVITY_VARIANT,".csv")),row.names=FALSE,na="")
print(head(comparison))
```
