**01：读取新输入，统一编码，并分别准备分类与联合比率数据**

先完成《00_开始与使用顺序》。本页代码在同一个 R 控制台会话中依次输入。代码块中的普通循环需要输入到最后一个右大括号才会执行；不要输入控制台显示的 `>` 或 `+` 提示符。

本步骤只读取你的输入文件，不修改它们。`Species` 仍表示一个物种及其在本分析中唯一的一套基因型。`observations.csv` 的每一行必须是一份可区分的独立实验组结果，而不是一篇文献中的重复转述。

这一整段完成读取、检查、编码和分析表构造。它允许同物种多条实验结果，并保留不能划入 High/Low 的区间。

<!-- R_BLOCK:01_PREPARE -->
```r
# 读取三份输入；空字段保留为缺失，绝不补成 0。
input_paths <- c(
  sites = file.path(INPUT_DIR, "sites.csv"),
  observations = file.path(INPUT_DIR, "observations.csv"),
  tree = file.path(INPUT_DIR, "tree.nwk")
)
if (any(!file.exists(input_paths))) stop("三份输入文件必须全部存在。")
# 先按字符读入，保留F/T残基和编号前导零；下方只显式转换数值列。
sites <- utils::read.csv(input_paths[["sites"]], check.names = FALSE,
  stringsAsFactors = FALSE, colClasses = "character", na.strings = c("", "NA"))
observations <- utils::read.csv(input_paths[["observations"]], check.names = FALSE,
  stringsAsFactors = FALSE, colClasses = "character", na.strings = c("", "NA"))
site_columns <- c("Site3", "Site20", "Site117", "Site151", "Site196", "Site315")
required_observation_columns <- c("RecordID", "ExperimentID", "Species", "Type",
  "Events", "Total", "Exact", "Lower", "Upper", "SourceID")
if (!all(c("Species", site_columns) %in% names(sites))) stop("sites.csv 缺少规定列。")
if (!all(required_observation_columns %in% names(observations))) {
  stop("observations.csv 缺少规定列，请对照输入格式说明。")
}
if (nrow(sites) != EXPECTED_SPECIES) stop("物种行数与 EXPECTED_SPECIES 不一致。")
if (nrow(observations) == 0L) stop("没有可用于分析的实验记录。")
sites$Species <- as.character(sites$Species)
observations$Species <- as.character(observations$Species)
if (anyNA(sites$Species) || any(trimws(sites$Species) == "") ||
    anyDuplicated(sites$Species)) stop("每个物种只能有一行非空的基因型信息。")
for (column in c("RecordID", "ExperimentID", "Species", "SourceID")) {
  value <- as.character(observations[[column]])
  if (anyNA(value) || any(trimws(value) == "")) stop(paste(column, "不能缺失。"))
  observations[[column]] <- value
}
if (anyDuplicated(observations$RecordID)) stop("RecordID 重复。")
if (anyDuplicated(observations$ExperimentID)) {
  stop("ExperimentID 重复：同一独立实验组结果不能重复进入模型。")
}
if (any(!observations$Species %in% sites$Species)) stop("有实验物种不在 sites.csv 中。")
observations$Type <- tolower(trimws(as.character(observations$Type)))
if (anyNA(observations$Type) || any(!observations$Type %in% c("count", "exact", "interval"))) {
  stop("Type 只能是 count、exact 或 interval。")
}

# 验证数字；例如 75% 应事先填成 0.75，而不是由程序猜测单位。
value_columns <- c("Events", "Total", "Exact", "Lower", "Upper")
for (column in value_columns) {
  raw <- observations[[column]]
  text <- trimws(as.character(raw))
  supplied <- !is.na(raw) & text != ""
  numeric_value <- suppressWarnings(as.numeric(text))
  if (any(supplied & (!is.finite(numeric_value) | is.na(numeric_value)))) {
    stop(paste(column, "有不能识别的数字。"))
  }
  observations[[column]] <- numeric_value
}
is_count <- observations$Type == "count"
is_exact <- observations$Type == "exact"
is_interval <- observations$Type == "interval"
if (any(is_count & (is.na(observations$Events) | is.na(observations$Total)))) {
  stop("count 行必须同时给出 Events 和 Total。")
}
if (any(is_count & (!is.na(observations$Exact) | !is.na(observations$Lower) |
                   !is.na(observations$Upper)))) stop("count 行不再重复填写比例或区间。")
if (any(is_count & (observations$Events < 0 | observations$Total < 1 |
    observations$Events > observations$Total |
    observations$Events != floor(observations$Events) |
    observations$Total != floor(observations$Total) |
    observations$Total > .Machine$integer.max))) stop("Events/Total 必须是合法整数计数。")
if (any(!is_count & (!is.na(observations$Events) | !is.na(observations$Total)))) {
  stop("分母未知的 exact/interval 行中，Events 和 Total 必须留空。")
}
if (any(is_exact & (is.na(observations$Exact) | observations$Exact <= 0 |
                   observations$Exact > 1))) {
  stop("未知分母的 exact 值必须在 (0,1]；本版不接受精确 0，也不会加减 epsilon。")
}
if (any(is_exact & (!is.na(observations$Lower) | !is.na(observations$Upper)))) {
  stop("exact 行的 Lower/Upper 必须留空。")
}
if (any(is_interval & (!is.na(observations$Exact) | is.na(observations$Lower) |
                      is.na(observations$Upper)))) stop("interval 行只填写 Lower 和 Upper。")
if (any(is_interval & (observations$Lower < 0 | observations$Upper > 1 |
                      observations$Lower >= observations$Upper))) {
  stop("区间必须满足 0 <= Lower < Upper <= 1。")
}

# 树已经定根；这里只读取和核对，不重新定根或修补枝长。
tree <- ape::read.tree(input_paths[["tree"]])
if (inherits(tree, "multiPhylo")) stop("tree.nwk 必须只有一棵树。")
if (anyDuplicated(tree$tip.label) || !setequal(tree$tip.label, sites$Species)) {
  stop("树末端与 sites.csv 的物种集合必须完全相同。")
}
if (!ape::is.rooted(tree)) stop("请提供已经确认根位置的树；本步骤不替你重新定根。")
if (is.null(tree$edge.length) || any(!is.finite(tree$edge.length)) ||
    any(tree$edge.length < 0)) stop("树必须有有效的非负枝长。")
A <- ape::vcv.phylo(tree, corr = TRUE)
A <- A[sites$Species, sites$Species, drop = FALSE]
if (any(!is.finite(A)) || max(abs(A - t(A))) > 1e-10) stop("树协方差无效。")
chol_A <- try(chol(A), silent = TRUE)
if (inherits(chol_A, "try-error")) stop("树协方差不是正定矩阵，请核对树；不会自动修补。")
L_A <- t(chol_A)

# 原始六个位点列保留；只对新增的建模编码统一 MISSING。
species <- sites
reference_residue <- c(Site3 = "C", Site20 = "K", Site117 = "K", Site151 = "C", Site196 = "C")
aa_allowed <- strsplit("ACDEFGHIKLMNPQRSTVWY", "")[[1L]]
encoding_dictionary <- list()
for (site in site_columns) {
  raw <- sites[[site]]
  residue <- toupper(trimws(as.character(raw)))
  residue[is.na(raw) | is.na(residue) |
    residue %in% c("", "NA", "X", "-", "MISSING", "INDEL")] <- "MISSING"
  if (any(!residue %in% c(aa_allowed, "MISSING"))) stop(paste(site, "存在非约定符号。"))
  if (site == "Site315") {
    m1 <- ifelse(residue == "MISSING", "MISSING",
      ifelse(residue %in% c("K", "T"), "K_or_T", ifelse(residue == "R", "R", "other")))
  } else {
    m1 <- ifelse(residue == "MISSING", "MISSING",
      ifelse(residue == reference_residue[[site]], paste0(reference_residue[[site]], "_validated"), "other"))
  }
  # validated 仅沿用原编码名称，不表示本程序验证过其功能。
  m2 <- ifelse(residue == "MISSING", "MISSING",
    ifelse(residue %in% c("C", "K", "S", "T"), "CKST", "non_CKST"))
  # M3按每个位点分别统计；365物种参考中的1%为3.65次，因此RARE_MIN=4。
  # MISSING单独保留；预测新增物种时沿用本次冻结的字典，不重新统计频数。
  frequencies <- table(residue[residue != "MISSING"])
  frequent <- names(frequencies)[frequencies >= RARE_MIN]
  m3 <- ifelse(residue == "MISSING", "MISSING", ifelse(residue %in% frequent, residue, "OTHER"))
  species[[paste0("M1_", site)]] <- m1
  species[[paste0("M2_", site)]] <- m2
  species[[paste0("M3_", site)]] <- m3
  encoding_dictionary[[site]] <- list(frequencies = frequencies, frequent = frequent)
}

# 单点或计数比率恰好0.5属于High；区间上界恰好0.5按约定属于Low。
observations$High <- NA_integer_
observations$BinaryReason <- "interval_crosses_threshold"
observations$High[is_count] <- as.integer(2 * observations$Events[is_count] >= observations$Total[is_count])
observations$High[is_exact] <- as.integer(observations$Exact[is_exact] >= 0.5)
observations$High[is_interval & observations$Lower >= 0.5] <- 1L
observations$High[is_interval & observations$Upper <= 0.5] <- 0L
observations$BinaryReason[!is.na(observations$High)] <- "classified"
observations$BinaryReason[is_interval & observations$Upper == 0.5] <- "interval_upper_50_LOW_by_rule"
observations$NoRatioInformation <- is_interval & observations$Lower == 0 & observations$Upper == 1
observations$NoRatioInformation[is.na(observations$NoRatioInformation)] <- FALSE
count_rows <- data.frame(Species = observations$Species,
  HighCount = as.integer(!is.na(observations$High) & observations$High == 1L),
  LowCount = as.integer(!is.na(observations$High) & observations$High == 0L),
  UnclassifiedCount = as.integer(is.na(observations$High)), RecordCount = 1L)
counts <- stats::aggregate(. ~ Species, data = count_rows, FUN = sum)
binary_counts <- data.frame(Species = species$Species, stringsAsFactors = FALSE)
for (column in setdiff(names(counts), "Species")) {
  value <- counts[[column]][match(species$Species, counts$Species)]
  value[is.na(value)] <- 0L
  binary_counts[[column]] <- as.integer(value)
}
binary_counts$Trials <- binary_counts$HighCount + binary_counts$LowCount
binary_counts$BinaryID <- paste0("binary:", binary_counts$Species)
if (!any(binary_counts$Trials > 0)) stop("没有可以进行独立分类分析的实验。")

# 候选结构保留十个：六个 M1/M2/M3 U/P，加上四个基线。
model_grid <- data.frame(
  Model = c("Null_U", "Phylogeny_only_P", "Site315_U", "Site315_P", "M1_U", "M1_P", "M2_U", "M2_P", "M3_U", "M3_P"),
  Group = rep(c("Null", "Site315", "M1", "M2", "M3"), each = 2L),
  Phylo = rep(c(FALSE, TRUE), 5L),
  Role = c(rep("baseline",4L), rep("primary",2L), rep("sensitivity",4L)),
  stringsAsFactors = FALSE)
predictor_groups <- list(Null = character(), Site315 = "M1_Site315",
  M1 = paste0("M1_",site_columns), M2 = paste0("M2_",site_columns), M3 = paste0("M3_",site_columns))
training_species <- list(binary = binary_counts$Species[binary_counts$Trials > 0],
  joint = unique(observations$Species[!observations$NoRatioInformation]))
if (!length(training_species$joint)) stop("全部区间都是 [0,1]，没有定量信息。")
blueprints <- list()
for (outcome in c("binary", "joint")) {
  index <- match(training_species[[outcome]], species$Species)
  for (group in names(predictor_groups)) {
    variables <- predictor_groups[[group]]
    categorical <- species[, variables, drop = FALSE]
    supported <- rep(TRUE, nrow(species))
    levels_used <- list()
    for (variable in variables) {
      observed_levels <- sort(unique(as.character(species[[variable]][index])))
      panel_levels <- c(observed_levels, setdiff(sort(unique(as.character(species[[variable]]))), observed_levels))
      categorical[[variable]] <- factor(species[[variable]], levels = panel_levels)
      levels_used[[variable]] <- observed_levels
      supported <- supported & species[[variable]] %in% observed_levels
    }
    if (length(variables)) {
      combination <- do.call(paste, c(species[,variables,drop=FALSE], sep="|"))
    } else {
      combination <- rep("(intercept-only)", nrow(species))
    }
    # 整个面板也只有一种水平的列，不能传给 model.matrix 的因子对比。
    matrix_variables <- names(categorical)[vapply(categorical,nlevels,integer(1)) >= 2L]
    if (length(matrix_variables)) {
      formula_x <- stats::reformulate(matrix_variables)
      raw_x <- stats::model.matrix(formula_x, data = categorical)
    } else {
      raw_x <- matrix(1, nrow(species), 1L, dimnames=list(NULL,"(Intercept)"))
    }
    keep <- rep(FALSE,ncol(raw_x))
    for (j in seq_len(ncol(raw_x))) {
      keep[j] <- colnames(raw_x)[j] != "(Intercept)" && length(unique(raw_x[index,j])) >= 2L
    }
    x <- raw_x[,keep,drop=FALSE]
    original_terms <- colnames(x)
    if (ncol(x)) {
      # 中心化只使用固定物种面板的基因型，不使用响应值。
      centers <- colMeans(x)
      x <- sweep(x,2L,centers,FUN="-")
    } else centers <- numeric()
    colnames(x) <- sprintf("x%02d",seq_len(ncol(x)))
    x_observed <- cbind(Intercept=1,x[index,,drop=FALSE])
    nonest <- estimability::nonest.basis(x_observed)
    estimable <- estimability::is.estble(cbind(Intercept=1,x),nonest)
    blueprints[[paste(outcome,group,sep="_")]] <- list(X=x,
      columns=data.frame(Column=colnames(x),Term=original_terms,Center=centers),
      variables=variables, levels=levels_used, supported=supported,
      observed_combination=combination %in% combination[index],
      linear_estimable=estimable, rank=qr(x_observed)$rank,
      parameter_columns=ncol(x_observed), observed_species=species$Species[index])
  }
}

input_hashes <- character(length(input_paths)); names(input_hashes) <- names(input_paths)
for (name in names(input_paths)) input_hashes[[name]] <- digest::digest(file=input_paths[[name]],algo="sha256")
prepared <- list(version=MODEL_VERSION,
  classification_rule="point_ge50_HIGH;interval_upper_le50_LOW;interval_lower_ge50_HIGH;strict_cross50_excluded",
  species=species,observations=observations,
  binary_counts=binary_counts,model_grid=model_grid,blueprints=blueprints,
  encoding_dictionary=encoding_dictionary,A=A,L_A=L_A,tree=tree,
  input_paths=input_paths,input_hashes=input_hashes,rare_min=RARE_MIN,seed=SEED)
print(table(observations$Type))
print(colSums(binary_counts[,c("HighCount","LowCount","UnclassifiedCount")]))
print(data.frame(Group=names(predictor_groups),
  Rank=vapply(blueprints[paste0("joint_",names(predictor_groups))], `[[`, integer(1), "rank"),
  Columns=vapply(blueprints[paste0("joint_",names(predictor_groups))], `[[`, integer(1), "parameter_columns")))
```
