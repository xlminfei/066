**08：可选地核对完整树矩阵与对应子矩阵的数值一致性**

这一页不需要第二棵树，也不重新定根。它只把同一张A中有来源记录的物种对应行列取出，与完整A的P模型进行比较。若这些物种已经等于全部物种，就没有较小的矩阵，本检查不适用。

先在05读取你要检查的路线并完成诊断；联合路线还须在当前R会话编译03的同一份模型。选择一个P模型：

<!-- R_BLOCK:08_PREPARE -->
```r
CHECK_MODEL <- "M1_P"
if(!CHECK_MODEL %in% names(bundles))stop("请先在05读取该模型。")
full_bundle<-bundles[[CHECK_MODEL]]

if(OUTCOME=="binary") {
  observed_species<-unique(as.character(full_bundle$request$training$Species))
} else {
  observed_species<-unique(as.character(full_bundle$request$source_rows$Species))

observed_index<-match(observed_species,full_bundle$species)
A_observed<-full_bundle$request$A[observed_species,observed_species,drop=FALSE]
if(!isTRUE(all.equal(A_observed,prepared$A[observed_species,observed_species,drop=FALSE],tolerance=0))) {
  stop("子矩阵不是来自当前同一张完整A。")
}
CHECK_SEED <- full_bundle$request$sampling$seed + 100001L
CHECK_CHAINS <- full_bundle$request$sampling$chains

CHECK_ITER <- full_bundle$request$sampling$iter

check_request<-list(parent_key=full_bundle$key,species=observed_species,A=A_observed,seed=CHECK_SEED,
  sampling=list(chains=CHECK_CHAINS,cores=CHECK_CORES,iter=CHECK_ITER,warmup=CHECK_WARMUP,
    adapt_delta=CHECK_ADAPT_DELTA,max_treedepth=CHECK_MAX_TREEDEPTH))
```

运行对应的子矩阵拟合。它单独保存，不会取代07使用的完整矩阵模型。中心化、位点编码、观测记录与先验都保持一致；随机种子可以不同。

```r
if(file.exists(sub_file)) {

    }
```

最后比较共同参数及来源物种的期望量。先确认子拟合本身的抽样质量；大MCSE不能被用来制造“通过”。

<!-- R_BLOCK:08_COMPARE -->
```r
pair_draws<-list();pair_sampler<-list();pair_quality<-list()
for(which_fit in c("full","sub")) {
  fit_here<-if(which_fit=="full")full_bundle$fit else sub_bundle$fit
  if(OUTCOME=="binary") {
    sf<-fit_here$fit
    da<-brms::as_draws_array(fit_here)
    vn<-posterior::variables(da)
    common<-posterior::subset_draws(da,variable=vn[grepl("^(b_|sd_|phi$)",vn)])
    nd<-full_bundle$training_data
    nd$Trials<-1L
    nd$Species<-factor(as.character(nd$Species),levels=if(which_fit=="full")full_bundle$species else observed_species)
    pred<-brms::posterior_epred(fit_here,newdata=nd,re_formula=NULL,allow_new_levels=FALSE)
    pd<-array(pred,dim=c(dim(da)[1],dim(da)[2],ncol(pred)))
    dimnames(pd)<-list(NULL,NULL,paste0("Expected[",as.character(nd$Species),"]"))
  } else {
    sf<-fit_here
    da<-posterior::as_draws_array(rstan::extract(sf,permuted=FALSE,inc_warmup=FALSE))
    vn<-posterior::variables(da)
    common<-posterior::subset_draws(da,variable=vn[grepl("^(alpha$|beta\\[|sd_phylo\\[|rho$|log_phi_ratio$|log_phi_count\\[)",vn)])
    ma<-rstan::extract(sf,pars="m",permuted=FALSE,inc_warmup=FALSE)
    use<-if(which_fit=="full")observed_index else seq_along(observed_species)
    pd<-ma[,,use,drop=FALSE]
    dimnames(pd)<-list(NULL,NULL,paste0("Expected[",observed_species,"]"))
  }
  pair_draws[[which_fit]]<-posterior::bind_draws(common,posterior::as_draws_array(pd),along="variable")
  pair_sampler[[which_fit]]<-rstan::get_sampler_params(sf,inc_warmup=FALSE)
  qa<-posterior::as_draws_array(rstan::extract(sf,permuted=FALSE,inc_warmup=FALSE))
  qv<-posterior::variables(qa)
  relevant<-if(OUTCOME=="binary")grepl("^(b_|sd_|r_|z_|phi$)",qv) else
    grepl("^(alpha$|beta\\[|z_phylo\\[|sd_phylo\\[|rho$|log_phi_ratio$|log_phi_count\\[|m\\[)",qv)
  pair_quality[[which_fit]]<-as.data.frame(posterior::summarise_draws(
    posterior::subset_draws(qa,variable=qv[relevant]),"rhat","ess_bulk","ess_tail"))
}
summaries<-list()
for(which_fit in names(pair_draws)) {
  ss<-as.data.frame(posterior::summarise_draws(pair_draws[[which_fit]],"mean","mcse_mean","rhat","ess_bulk","ess_tail"))
  sp<-pair_sampler[[which_fit]];dv<-td<-0L;bf<-numeric(length(sp))
  quality<-pair_quality[[which_fit]]
  depth_limit<-if(which_fit=="full")full_bundle$request$sampling$max_treedepth else sub_bundle$request$sampling$max_treedepth
  for(ch in seq_along(sp)) {
    dv<-dv+sum(sp[[ch]][,"divergent__"])
    td<-td+sum(sp[[ch]][,"treedepth__"]>=depth_limit)
    bf[ch]<-mean(diff(sp[[ch]][,"energy__"])^2)/stats::var(sp[[ch]][,"energy__"])
  }
  if(any(!is.finite(as.matrix(ss[,c("mean","mcse_mean","rhat","ess_bulk","ess_tail")])) ) ||
     any(!is.finite(as.matrix(quality[,c("rhat","ess_bulk","ess_tail")]))) ||
     max(quality$rhat)>=MAX_RHAT || min(quality$ess_bulk)<MIN_ESS || min(quality$ess_tail)<MIN_ESS ||
     any(ss$mcse_mean<=0) || max(ss$rhat)>=MAX_RHAT || min(ss$ess_bulk)<MIN_ESS ||
     min(ss$ess_tail)<MIN_ESS || dv>0 || td>0 || any(!is.finite(bf)) || min(bf)<=.3) {
    stop("完整或子矩阵拟合需要进一步诊断，不能把数值比较写成通过。")
  }
  summaries[[which_fit]]<-ss
}
f<-summaries$full;s<-summaries$sub
if(!setequal(f$variable,s$variable))stop("两次拟合的共同量不同。")
s<-s[match(f$variable,s$variable),]
comparison<-data.frame(Quantity=f$variable,FullMean=f$mean,SubmatrixMean=s$mean,
  Difference=f$mean-s$mean,JointMCSE=sqrt(f$mcse_mean^2+s$mcse_mean^2))
comparison$ReviewFlag<-ifelse(abs(comparison$Difference)<=2*comparison$JointMCSE,
  "within_2_MCSE","review_difference")
utils::write.csv(comparison,file.path(CHECK_DIR,paste0(OUTCOME,"_",CHECK_MODEL,"_comparison.csv")),row.names=FALSE)
print(comparison)
```

`within_2_MCSE`仅表示本次两次模拟的均值差异在这一数值尺度内，不是统计等价证明，也不能证明模型具有生物学解释或可靠的外推能力。出现`review_difference`时应结合采样精度、参数数量和预测差异检查；不能为了通过而增大阈值。

高斯随机效应的边缘协方差就是原矩阵的对应子矩阵。因此在观测、先验和设计矩阵相同时，两种写法对共同量具有相同的目标分布。有限抽样的均值仍会不同；同时检查多个相关量时，也可能有少量结果越过2倍MCSE。应先排查目标和数据是否一致，再解释模拟差异，不能反复更换种子直到所有标记消失。

如子矩阵拟合需要更多采样，调整本页的`CHECK_ITER`、`CHECK_WARMUP`或`CHECK_ADAPT_DELTA`后，从`check_request`开始重新执行准备和拟合。新设置使用新的缓存文件，不需要改动已经通过诊断的完整矩阵拟合。
