**07：输出High概率，以及整体期望比率的后验区间**

先对binary和joint两条路线分别完成05的诊断。这里保留M1/M2/M3的U/P共六个模型和全部目标物种。基线用于比较，不混入六模型的正式长表。

本页的比率区间只来自每个物种整体期望`m`的后验抽样；不模拟未来实验，不构造未来分母。点估计沿用原流程的后验中位数，区间取2.5%和97.5%分位数。

<!-- R_BLOCK:07_PREDICT -->
```r
VARIANT <- "primary"
prediction_models <- c("M1_U","M1_P","M2_U","M2_P","M3_U","M3_P")
prepared<-readRDS(file.path(RUN_DIR,"prepared.rds"))
if(!identical(prepared$version,MODEL_VERSION))stop("准备结果使用了旧分类版本，请重新准备。")
for(name in names(prepared$input_paths)) {
  if(!identical(digest::digest(file=prepared$input_paths[[name]],algo="sha256"),prepared$input_hashes[[name]])) {
    stop("输入发生改变，请不要使用旧拟合预测。")
  }
}
ix<-utils::read.csv(file.path(RUN_DIR,"fit_index.csv"),stringsAsFactors=FALSE)
result_rows<-vector("list",length(prediction_models))
expected_draws<-list(binary=list(),joint=list())
selection_used<-list()
joint_code_seen<-NULL
for(j in seq_along(prediction_models)) {
  model<-prediction_models[j]
  answer<-data.frame(Species=prepared$species$Species,Model=model,
    Role=prepared$model_grid$Role[match(model,prepared$model_grid$Model)],
    PrHigh=NA_real_,PrHighLower95=NA_real_,PrHighUpper95=NA_real_,
    PredictedRatio=NA_real_,RatioLower95=NA_real_,RatioUpper95=NA_real_,stringsAsFactors=FALSE)
  for(outcome in c("binary","joint")) {
    sel<-ix[ix$Outcome==outcome & ix$Variant==VARIANT & ix$Model==model,,drop=FALSE]
    if(!nrow(sel))stop(paste("缺少",outcome,model,"的拟合。"))
    sel<-sel[nrow(sel),,drop=FALSE]
    path<-file.path(RUN_DIR,sel$RelativeFile)
    if(!identical(digest::digest(file=path,algo="sha256"),sel$FileSHA256))stop("拟合文件哈希改变。")
    obj<-readRDS(path)
    if(!identical(obj$request$version,MODEL_VERSION))stop("拟合来自旧运行版本，不能用于本版预测。")
    if(!identical(obj$key,sel$Key) || !identical(obj$key,digest::digest(obj$request,algo="sha256")) ||
       !identical(obj$request$source_hashes,prepared$input_hashes) || !identical(obj$species,prepared$species$Species)) {
      stop("预测对象与当前输入/物种顺序不一致。")
    }
    if(obj$request$spec$Phylo && !isTRUE(all.equal(obj$request$A,prepared$A,tolerance=0)))stop("拟合采用的树矩阵已经不同。")
    dg<-utils::read.csv(file.path(RUN_DIR,"results",paste0("diagnostics_",outcome,"_",VARIANT,".csv")),stringsAsFactors=FALSE)
    dg<-dg[dg$Model==model & dg$Key==obj$key,,drop=FALSE]
    if(nrow(dg)!=1L || dg$Status!="PASS")stop("当前选中的拟合尚未通过05诊断，请刷新诊断。")
    bp<-obj$blueprint
    group<-obj$request$spec$Group
    if(!identical(bp,prepared$blueprints[[paste0(outcome,"_",group)]]))stop("模型编码/支持状态与当前准备结果不一致。")
    supported<-bp$supported
    if(outcome=="binary") {
      nd<-cbind(data.frame(Species=factor(obj$species,levels=obj$species),Trials=1L),as.data.frame(bp$X))
      md<-brms::posterior_epred(obj$fit,newdata=nd,re_formula=NULL,allow_new_levels=FALSE)
      if(ncol(md)!=length(obj$species))stop("分类预测物种数不一致。")
      has_data<-prepared$binary_counts$Trials>0
    } else {
      if(is.null(joint_code_seen))joint_code_seen<-obj$request$code_hash
      if(!identical(joint_code_seen,obj$request$code_hash))stop("六个联合模型的观测模型代码不同，不能静默混用。")
      ma<-rstan::extract(obj$fit,pars="m",permuted=FALSE,inc_warmup=FALSE)
      md<-matrix(ma,nrow=dim(ma)[1]*dim(ma)[2],ncol=dim(ma)[3])
      if(ncol(md)!=length(obj$species))stop("期望比率物种数不一致。")
      has_data<-obj$species %in% prepared$observations$Species[!prepared$observations$NoRatioInformation]
    }
    if(any(!is.finite(md)) || any(md<0 | md>1))stop("期望量预测不在有效范围内。")
    med<-low<-high<-rep(NA_real_,length(obj$species))
    for(s in which(supported)) {
      med[s]<-stats::median(md[,s])
      q<-stats::quantile(md[,s],probs=c(.025,.975),names=FALSE)
      low[s]<-q[1];high[s]<-q[2]
    }
    note<-ifelse(!supported,"unseen_level_no_numeric_prediction",
      ifelse(!bp$linear_estimable,"prior_dependent_unidentified_direction",
        ifelse(bp$observed_combination,"observed_combination","new_combination")))
    prefix<-if(outcome=="binary")"Binary"else"Ratio"
    answer[[paste0(prefix,"DataStatus")]]<-ifelse(has_data,"observed","unlabelled_projection")
    answer[[paste0(prefix,"Supported")]]<-supported
    answer[[paste0(prefix,"FixedEffectEstimable")]]<-bp$linear_estimable
    answer[[paste0(prefix,"CombinationSeen")]]<-bp$observed_combination
    answer[[paste0(prefix,"PredictionNote")]]<-note
    if(outcome=="binary") {
      answer$PrHigh<-med;answer$PrHighLower95<-low;answer$PrHighUpper95<-high
    } else {
      answer$PredictedRatio<-med;answer$RatioLower95<-low;answer$RatioUpper95<-high
    }
    md[,!supported]<-NA_real_
    colnames(md)<-obj$species
    expected_draws[[outcome]][[model]]<-md
    selection_used[[paste(outcome,model)]]<-sel
  }
  missing_matrix<-as.matrix(prepared$species[,paste0("M1_",c("Site3","Site20","Site117","Site151","Site196","Site315"))])
  answer$AnyMissingSite<-rowSums(missing_matrix=="MISSING")>0
  result_rows[[j]]<-answer
}
long_output<-do.call(rbind,result_rows);rownames(long_output)<-NULL
if(nrow(long_output)!=nrow(prepared$species)*6L || anyDuplicated(paste(long_output$Species,long_output$Model))) {
  stop("预测长表行数或唯一性不正确。")
}
wide_output<-data.frame(Species=prepared$species$Species,stringsAsFactors=FALSE)
for(model in prediction_models) {
  part<-long_output[long_output$Model==model,,drop=FALSE]
  part<-part[match(wide_output$Species,part$Species),setdiff(names(part),c("Species","Model")),drop=FALSE]
  names(part)<-paste0(model,"_",names(part))
  wide_output<-cbind(wide_output,part)
}
utils::write.csv(long_output,file.path(RUN_DIR,"results","species_predictions_long.csv"),row.names=FALSE,na="")
utils::write.csv(wide_output,file.path(RUN_DIR,"results","species_predictions_wide.csv"),row.names=FALSE,na="")
saveRDS(list(draws=expected_draws,selection=selection_used,quantity="posterior expected value",point_summary="median"),
  file.path(RUN_DIR,"results","expected_value_draws.rds"))
print(head(long_output))
```

`Supported`只表示编码水平可以被当前模型处理，不等于科学证据充分。M3的新组合即使每个单独水平都见过，也可能落在训练矩阵无法识别的方向；相应标记会保留。

没有见过的水平保留该物种行，数字留空，不把它自动替换成某个参考水平后给出貌似完整的结果。同一模型内各物种共享后验抽样，保存的抽样矩阵保留了这种关系；不要把365个预测点当作365份新实验。
